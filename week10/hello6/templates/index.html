from flask import Flask, render_template, request app = Flask(__name__) # Route to show the form @app.route("/") def index(): return render_template("index.html") # Route that the form 'action' points to @app.route("/greet") def greet(): # Gets the name from the form input name = request.args.get("name", "world") return render_template("greet.html", name=name)
